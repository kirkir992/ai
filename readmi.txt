Używaj kodu z rozwagą.Krok 3: Konfiguracja Nginx do obsługi StreamlitaMusimy powiedzieć Nginxowi, aby chronił dostęp i przekazywał ruch do naszej aplikacji.Otwórz domyślny plik konfiguracyjny Nginx:bashnano /etc/nginx/sites-available/default
Używaj kodu z rozwagą.Skasuj jego zawartość i wklej poniższy, gotowy blok (zoptymalizowany pod protokół WebSocket, którego używa Streamlit):nginxserver {
    listen 80 default_server;
    listen [::]:80 default_server;

    server_name _;

    # Włączenie ochrony hasłem
    auth_basic "AI Portal - Access Restricted";
    auth_basic_user_file /etc/nginx/.htpasswd;

    location / {
        proxy_pass http://127.0.0.1:8501;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Kluczowe dla Streamlita: obsługa WebSockets
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 86400;
    }
}
